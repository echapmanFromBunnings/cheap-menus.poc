<script>
  // Parse URL parameters
  const urlParams = new URLSearchParams(window.location.search);
  const kioskMode = urlParams.get('kiosk') === 'true';
  
  // Rotation state
  let rotation = 0;
  
  // Get elements
  const menuImage = document.getElementById('menuImage');
  const navOverlay = document.getElementById('navOverlay');
  const toggleButtonsBtn = document.getElementById('toggleButtonsBtn');
  const rotateBtn = document.getElementById('rotateBtn');
  
  // Kiosk mode: hide buttons by default
  if (kioskMode) {
    navOverlay.classList.add('hidden');
  }
  
  // Rotate button handler
  rotateBtn.addEventListener('click', function() {
    rotation = (rotation + 90) % 360;
    menuImage.style.transform = 'rotate(' + rotation + 'deg)';
  });
  
  // Toggle buttons visibility
  toggleButtonsBtn.addEventListener('click', function() {
    navOverlay.classList.toggle('hidden');
    if (navOverlay.classList.contains('hidden')) {
      toggleButtonsBtn.textContent = 'Show Buttons';
    } else {
      toggleButtonsBtn.textContent = 'Hide Buttons';
    }
  });
  
  // Update toggle button text based on initial state
  if (navOverlay.classList.contains('hidden')) {
    toggleButtonsBtn.textContent = 'Show Buttons';
  } else {
    toggleButtonsBtn.textContent = 'Hide Buttons';
  }
</script>
