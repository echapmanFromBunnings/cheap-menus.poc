<script>
  // Parse URL parameters
  const urlParams = new URLSearchParams(window.location.search);
  const kioskMode = urlParams.get('kiosk') === 'true';
  const rotationParam = urlParams.get('rotation');
  const transitionParam = urlParams.get('transition');
  
  // Current page number (5-8)
  const currentPage = 6;
  const pages = [5, 6, 7, 8];
  
  // Parse transition times (only if parameter exists)
  let transitionTimes = [];
  if (transitionParam) {
    transitionTimes = transitionParam.split(',').map(t => parseInt(t.trim(), 10)).filter(t => !isNaN(t) && t > 0);
  }
  
  // Rotation state
  let rotation = rotationParam ? parseInt(rotationParam, 10) : 0;
  
  // Get elements
  const menuImage = document.getElementById('menuImage');
  const navOverlay = document.getElementById('navOverlay');
  const toggleButtonsBtn = document.getElementById('toggleButtonsBtn');
  const fitButtonsContainer = document.getElementById('fitButtonsContainer');
  const containBtn = document.getElementById('containBtn');
  const coverBtn = document.getElementById('coverBtn');
  const fillBtn = document.getElementById('fillBtn');
  const rotateBtn = document.getElementById('rotateBtn');
  const timerDisplay = document.getElementById('timerDisplay');
  const nextButton = document.getElementById('nextButton');
  
  // Kiosk mode: hide buttons by default
  if (kioskMode) {
    navOverlay.classList.add('hidden');
    timerDisplay.classList.add('hidden');
    nextButton.classList.add('hidden');
  }
  
  // Hide timer and next button if no transition parameter
  if (!transitionParam) {
    timerDisplay.classList.add('hidden');
    nextButton.classList.add('hidden');
  }
  
  // Object-fit mode handlers
  function setObjectFit(mode, activeButton) {
    menuImage.style.objectFit = mode;
    // Remove active class from all buttons
    containBtn.classList.remove('active');
    coverBtn.classList.remove('active');
    fillBtn.classList.remove('active');
    // Add active class to clicked button
    activeButton.classList.add('active');
  }
  
  containBtn.addEventListener('click', function() {
    setObjectFit('contain', containBtn);
  });
  
  coverBtn.addEventListener('click', function() {
    setObjectFit('cover', coverBtn);
  });
  
  fillBtn.addEventListener('click', function() {
    setObjectFit('fill', fillBtn);
  });
  
  // Apply initial rotation if specified
  if (rotation !== 0) {
    updateImageTransform();
  }
  
  // Auto-rotation functionality
  let currentIndex = pages.indexOf(currentPage);
  let timeRemaining = transitionTimes[currentIndex];
  let timerInterval;
  
  function updateTimer() {
    if (!kioskMode) {
      timerDisplay.textContent = 'Next: ' + timeRemaining + 's';
    }
  }
  
  function buildUrl(pageNum) {
    const url = new URL(window.location.origin + '{{ site.baseurl }}/' + pageNum + '.html');
    if (kioskMode) {
      url.searchParams.set('kiosk', 'true');
    }
    if (rotation !== 0) {
      url.searchParams.set('rotation', rotation.toString());
    }
    url.searchParams.set('transition', transitionTimes.join(','));
    return url.toString();
  }
  
  function startAutoRotation() {
    updateTimer();
    
    timerInterval = setInterval(function() {
      timeRemaining--;
      updateTimer();
      
      if (timeRemaining <= 0) {
        // Clear interval before navigation
        clearInterval(timerInterval);
        // Move to next page
        currentIndex = (currentIndex + 1) % pages.length;
        const nextPage = pages[currentIndex];
        window.location.href = buildUrl(nextPage);
      }
    }, 1000);
  }
  
  // Start auto-rotation only if transition parameter exists
  if (transitionParam) {
    startAutoRotation();
  }
  
  // Function to update image transform with rotation and scaling
  function updateImageTransform() {
    var transform = 'rotate(' + rotation + 'deg)';
    
    // When rotated 90 or 270 degrees, we need to scale the image
    // to ensure it fills the container properly
    if (rotation === 90 || rotation === 270) {
      var container = document.querySelector('.fullscreen-container');
      var containerWidth = container.clientWidth;
      var containerHeight = container.clientHeight;
      var aspectRatio = containerWidth / containerHeight;
      
      // Scale to ensure the rotated image fills the container
      var scale = aspectRatio;
      transform += ' scale(' + scale + ')';
    }
    
    menuImage.style.transform = transform;
  }
  
  // Rotate button handler
  rotateBtn.addEventListener('click', function() {
    rotation = (rotation + 90) % 360;
    updateImageTransform();
  });
  
  // Toggle buttons visibility
  toggleButtonsBtn.addEventListener('click', function() {
    navOverlay.classList.toggle('hidden');
    nextButton.classList.toggle('hidden');
    fitButtonsContainer.classList.toggle('hidden');
    if (navOverlay.classList.contains('hidden')) {
      toggleButtonsBtn.textContent = 'Show Buttons';
    } else {
      toggleButtonsBtn.textContent = 'Hide Buttons';
    }
  });
  
  // Next button handler
  nextButton.addEventListener('click', function() {
    currentIndex = (currentIndex + 1) % pages.length;
    const nextPage = pages[currentIndex];
    window.location.href = buildUrl(nextPage);
  });
  
  // Update toggle button text based on initial state
  if (navOverlay.classList.contains('hidden')) {
    toggleButtonsBtn.textContent = 'Show Buttons';
  } else {
    toggleButtonsBtn.textContent = 'Hide Buttons';
  }
  
  // Update navigation links to preserve URL parameters
  const navLinks = navOverlay.querySelectorAll('a[href*="5.html"], a[href*="6.html"], a[href*="7.html"], a[href*="8.html"]');
  navLinks.forEach(function(link) {
    const pageMatch = link.href.match(/(\d+)\.html/);
    if (pageMatch) {
      link.href = buildUrl(parseInt(pageMatch[1]));
    }
  });
</script>
